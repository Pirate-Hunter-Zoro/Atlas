\begin{column}{0.54\textwidth}
\small
\begin{itemize}\setlength{\itemsep}{0.45em}
\item \textbf{Attention}: each token looks back over the earlier tokens and takes in what it needs from them.
\item \textbf{KV cache}: the per-turn record of every earlier token's attention keys and values, kept so they are not recomputed.
\item \textbf{Expert}: one routed feed-forward network --- its own weight matrices (gate, up, down).
\item \textbf{Router}: scores all 256 experts in a block from the token's hidden state --- the vector that block computed --- and keeps the top $k=8$. Those eight picks are the token's \emph{route} through the block.
\end{itemize}
\end{column}
\end{columns}

\begin{frame}{Batching barely pays at $t=2$, and pays at large $t$}

\begin{columns}[T]
\begin{column}{0.50\textwidth}
\footnotesize
\setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}
$n$ = experts in a block, $k$ = picked per token, $t$ = tokens sharing one pass.

\smallskip
Assumes: tokens pick experts independently (naive); all within \emph{one} block; each block has its \emph{own} 256 experts, so $19{,}200 = 256 \times 75$ experts in all.

\smallskip
$I_i = 1$ when expert $i$ is used by any token; $U = \sum_{i=1}^{n} I_i$.
\begin{align*}
E[U] &= E\Bigl[\textstyle\sum_{i=1}^{n} I_i\Bigr] = n\,E[I_1] \quad\text{(experts alike)}\\
E[I_1] &= 1 - P(\text{no token uses expert 1})\\
       &= 1 - \Bigl(\textstyle\prod_{i=1}^{k} \tfrac{n-i}{n-i+1}\Bigr)^{t} = 1 - \bigl(\tfrac{n-k}{n}\bigr)^{t}\\
\Rightarrow\; E[U] &= n\Bigl(1-\bigl(\tfrac{n-k}{n}\bigr)^{t}\Bigr)
\end{align*}
GLM: $n=256$, $k=8$, so $E[U] = 256\bigl(1-(31/32)^{t}\bigr)$.
\end{column}
\begin{column}{0.48\textwidth}
\centering
\begin{tikzpicture}
\begin{axis}[

\begin{frame}[fragile]{Starting Colibri: \texttt{slurm\_jobs/colibri\_serve.sbatch}}

\begin{columns}[T]
\begin{column}{0.47\textwidth}
\begin{lstlisting}
#SBATCH --partition=c3_short
#SBATCH --ntasks=1
#SBATCH --gres=gpu:1
#SBATCH --cpus-per-task=80
#SBATCH --mem=800G
#SBATCH --time=0-08:00:00
\end{lstlisting}
\begin{lstlisting}
SERVE=(
  python3 "$COLI_LAUNCHER" serve
  --model "$COLI_MODEL"
  --model-id "$COLI_MODEL_ID"
  --host 127.0.0.1 --port "$COLI_PORT"
  --gpu auto
  --auto-tier
  --max-queue "${COLI_MAX_QUEUE:-8}"
  --queue-timeout "${COLI_QUEUE_TIMEOUT:-3600}"
)
SERVE+=(--ctx "${COLI_CTX:-131072}")
\end{lstlisting}
\end{column}
\begin{column}{0.51\textwidth}
\scriptsize
\begin{itemize}\setlength{\itemsep}{0.15em}
\item \texttt{c3\_short}: compute300--305, the shallower queue.
\item \texttt{gpu:1}: one A40; four are worth 0.9\,\%.
\item \texttt{80} CPUs: the expert tail runs on the CPU.
\item \texttt{800G}: room to pin the expert bank in RAM.
\item \texttt{-{}-gpu auto}: turns the GPU on; with none of \texttt{-{}-gpu}, \texttt{-{}-vram} or \texttt{-{}-auto-tier} the engine runs CPU-only and says nothing.
\item \texttt{-{}-auto-tier}: applies the VRAM/RAM split \texttt{coli doctor} plans --- what sits on the GPU.
\item \texttt{-{}-max-queue}: requests that may wait; one is served at a time.
\item \texttt{-{}-queue-timeout}: an agent's long first prefill is not refused at the 300\,s default.
\item \texttt{-{}-ctx}: the context window in tokens; it sets the KV cache.
\end{itemize}
\end{column}
\end{columns}

\end{frame}
