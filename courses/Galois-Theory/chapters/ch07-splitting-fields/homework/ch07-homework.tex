\noindent\textbf{Standing assumptions.} From here on we assume the fundamental theorem: for a finite Galois extension $K/F$, the maps $H\mapsto K^H$ and $L\mapsto \operatorname{Gal}(K/L)$ are mutually inverse, inclusion-reversing bijections between the subgroups of $\operatorname{Gal}(K/F)$ and the intermediate fields of $K/F$, with $[K:K^H]=|H|$; and $L/F$ is Galois exactly when $\operatorname{Gal}(K/L)$ is normal in $\operatorname{Gal}(K/F)$, in which case $\operatorname{Gal}(L/F)\cong \operatorname{Gal}(K/F)/\operatorname{Gal}(K/L)$. As before, $\zeta_n=e^{2\pi i/n}$, and all fields are subfields of $\CC$ unless stated otherwise.

\begin{problem}{13}
  \textit{The subfields of $\QQ(\zeta_p)$}\par
  Let $p$ be an odd prime, $\zeta=\zeta_p$, $K=\QQ(\zeta)$ and $G=\operatorname{Gal}(K/\QQ)$.
  \begin{enumerate}[label=(\alph*)]
    \item Show that $G$ is cyclic of order $p-1$.
    \item Show that for each divisor $d$ of $p-1$ there is exactly one intermediate field $K_d$ with $[K_d:\QQ]=d$, and that every such $K_d$ is Galois over $\QQ$ with cyclic Galois group.
    \item Take $p=13$. Check that $2$ is a primitive root modulo $13$, and use it to list every subgroup of $(\ZZ/13\ZZ)^\times$ together with the degree over $\QQ$ of the corresponding fixed field.
    \item Let $H\le G$ be the subgroup of index $d$, viewed inside $(\ZZ/p\ZZ)^\times$, and set $\eta_H=\sum_{a\in H}\zeta^a$ (a \emph{Gaussian period}). Show that $K_d=\QQ(\eta_H)$. \emph{Hint:} $\zeta,\zeta^2,\dots,\zeta^{p-1}$ is a $\QQ$-basis of $K$.
  \end{enumerate}
\end{problem}
% ===== SOLUTION 13 =====
\noindent\textbf{(a)}
\begin{proof}
The minimal polynomial of $\zeta$ over $\QQ$ is
\[ \Phi_p(x)=\sum_{i=1}^{p}x^{i-1}=x^{p-1}+\dots+x+1, \]
whose $p-1$ roots are $\zeta,\zeta^2,\zeta^3,\dots,\zeta^{p-1}$. Each $\sigma\in G$ sends $\zeta$ to a root of $\Phi_p$, and for each root $\zeta^k$, $k\in\{1,\dots,p-1\}$, there is exactly one $\sigma_k\in G$ with $\sigma_k(\zeta)=\zeta^k$. So there are $p-1$ of them, and $|G|=p-1$.

Since $(\ZZ/p\ZZ)^\times$ is cyclic, there is a primitive root $g$ modulo $p$. The automorphism $\sigma_g$ with $\sigma_g(\zeta)=\zeta^g$ satisfies $\sigma_g^{\,j}(\zeta)=\zeta^{g^j}$, and in particular $\sigma_g^{\,p-1}(\zeta)=\zeta^{g^{p-1}}=\zeta$ by Fermat's little theorem.

Suppose $\sigma_k\in G$, with $\sigma_k(\zeta)=\zeta^k$. Since $g$ is a primitive root modulo $p$, there is an $n$ with $g^n\equiv k\pmod p$. So $\sigma_g^{\,n}(\zeta)=\zeta^{g^n}=\zeta^k=\sigma_k(\zeta)$, and since an element of $G$ is determined by its value at $\zeta$, $\sigma_g^{\,n}=\sigma_k$. Hence $\sigma_g$ generates $G$, and $G$ is cyclic of order $p-1$.
\end{proof}
\noindent\textbf{(b)}
\begin{proof}
For a subgroup $H\le G$ the index is $[G:H]=|G|/|H|$, so the subgroup of order $(p-1)/d$ has index $d$.

For $H\le G$ the fixed field is $K^H=\{k\in K:\sigma(k)=k \text{ for every }\sigma\in H\}$, and $[K:K^H]=|H|$.

A cyclic group has exactly one subgroup of each order dividing its order. Since $G$ is cyclic, it has exactly one subgroup $H$ of order $(p-1)/d$. By the correspondence $H$ has exactly one fixed field $K^H=K_d$, and by the index computation above $[K_d:\QQ]=d$.

Since $G$ is cyclic, every subgroup is normal, so $H\unlhd G$. Hence $K_d=K^H$ is Galois over $\QQ$, with Galois group $G/H$, a quotient of the cyclic group $G$ and therefore cyclic.
\end{proof}

Since $3\in H$ and $H$ is a subgroup, multiplication by $3$ permutes $H$, so $\sigma_3(\eta_H)=\eta_H$. Since $2\notin H$, $\sigma_2(\eta_H)=\eta_N$: the exponents $2,6,8,18,20,24$ reduce to $2,6,8,5,7,11$, which is $N$. And $\sigma_2^2=\sigma_4$ with $4\in H$, so $\sigma_2^2(\eta_H)=\eta_H$. The stabilizer of $\eta_H$ is therefore exactly $H$, and its orbit is $\{\eta_H,\eta_N\}$.

The two periods are distinct. If $\eta_H=\eta_N$, the difference has coefficient $+1$ on $\zeta^a$ for $a\in H$ and $-1$ for $a\in N$; those twelve coefficients are rational and not all zero, so the difference cannot vanish, because $\zeta,\dots,\zeta^{12}$ is a $\QQ$-basis of $K$.

The concrete case also gives $\eta_H+\eta_N=-1$, since $H\cup N=\{1,\dots,12\}$ and $\sum_{i=1}^{12}\zeta^i=-1$, and $3N=N$ because $3\in H$. For the product, $m_0=0$ and $m_1=3$; the map $(a,b)\mapsto(2b,2a)$ sends $S_t$ one-to-one onto $S_{2t}$ with inverse $(c,d)\mapsto(7d,7c)$, so $m_{2t}=m_t$. Since $2$ has order $12$ mod $13$, every nonzero $t$ has $m_t=3$, and the $36$ pairs are accounted for. Hence $\eta_H\eta_N=3\sum_{i=1}^{12}\zeta^i=-3$, and $\eta_H,\eta_N$ are the roots of $x^2+x-3$.

\begin{proof}
Let $H\le G$ have index $d$ and put $\eta_H=\sum_{a\in H}\zeta^{a}$; for a coset $aH$ write $\eta_{aH}=\sum_{h\in H}\zeta^{ah}$, so that $\sigma_a(\eta_H)=\eta_{aH}$.

The periods $\eta_{aH}$, as $aH$ runs over $G/H$, are pairwise distinct. Suppose $\eta_{aH}=\eta_{bH}$ with $aH\ne bH$, and let
\[ P(x)=\sum_{h\in H}x^{ah}-\sum_{h\in H}x^{bh}. \]
Then $P(\zeta)=0$, and the coefficients of $P$ are $+1$ on the exponents in $aH$, $-1$ on those in $bH$, and $0$ elsewhere. Since $\Phi_p$ is the minimal polynomial of $\zeta$ over $\QQ$, $\Phi_p$ divides $P$; both have degree at most $p-1$, so $P=c\,\Phi_p$ for some constant $c$. The cosets $aH$ and $bH$ are disjoint and nonempty, so $P$ has a coefficient $+1$ and a coefficient $-1$; hence $P\ne0$ and $c\ne0$. But every coefficient of $c\,\Phi_p$ equals $c$, which cannot be both $+1$ and $-1$. So distinct cosets give distinct periods.

Let $m(x)=\prod_{aH\in G/H}(x-\eta_{aH})$. Its coefficients are the elementary symmetric functions of the periods. Each $\sigma\in G$ permutes the periods, so it fixes every coefficient, and the coefficients lie in $K^{G}$. Since $K/\QQ$ is Galois, $[K:K^{G}]=|G|=p-1$, so $K^{G}=\QQ$; hence $m\in\QQ[x]$, monic of degree $d$, with $m(\eta_H)=0$.

Suppose $m=fg$ with $f,g\in\QQ[x]$ monic of positive degree, and let $\alpha$ be a period with $f(\alpha)=0$. Every period is $\sigma(\alpha)$ for some $\sigma\in G$. A field automorphism fixing $\QQ$ commutes with evaluation of a rational-coefficient polynomial, so $f(\sigma(\alpha))=\sigma(f(\alpha))=0$: every period is a root of $f$. Hence $\deg f\ge d$, while $f\mid m$ gives $\deg f\le d$. So $f=m$ and $g$ is constant, and $m$ is irreducible over $\QQ$.

Therefore $[\QQ(\eta_H):\QQ]=d$. Since $\eta_H$ is fixed by $H$, $\QQ(\eta_H)\subseteq K^{H}=K_d$, and both have degree $d$ over $\QQ$, so $\QQ(\eta_H)=K_d$.
\end{proof}

\begin{problem}{14}

  \begin{enumerate}[label=(\alph*)]
    \item Show that $3$ is a primitive root modulo $7$, and deduce that $\operatorname{Gal}(K/\QQ)=\langle\sigma\rangle$ where $\sigma(\zeta)=\zeta^3$.
    \item Let $\alpha=\zeta+\zeta^2+\zeta^4$ and $\beta=\zeta^3+\zeta^5+\zeta^6$. Show that $\alpha+\beta=-1$ and $\alpha\beta=2$, and deduce that the unique quadratic subfield of $K$ is $\QQ(\sqrt{-7})$.
    \item Let $\eta=\zeta+\zeta^{-1}=2\cos(2\pi/7)$. Show that the minimal polynomial of $\eta$ over $\QQ$ is $x^3+x^2-2x-1$, and that $\QQ(\eta)$ is the unique cubic subfield of $K$.
    \item Draw the lattice of intermediate fields of $K/\QQ$, and beside it the lattice of subgroups of $\operatorname{Gal}(K/\QQ)$ drawn \emph{upside down} (trivial subgroup at the top). Label every edge with the degree or index it represents, and match each subgroup with its fixed field.
  \end{enumerate}
\end{problem}

\noindent\textbf{(a)}
\begin{proof}
The cyclic subgroup generated by each residue modulo $7$ is
\[
\langle 2\rangle=\{2,4,1\},\qquad \langle 3\rangle=\{3,2,6,4,5,1\},\qquad \langle 5\rangle=\{5,4,6,2,3,1\}.
\]
So $2$ has order $3$ and is not a primitive root, while $3$ and $5$ have order $6$.

Take $\sigma(\zeta)=\zeta^{3}$. Since $3$ has order $6$, the powers $\sigma,\sigma^{2},\dots,\sigma^{6}$ send $\zeta$ to $\zeta^{3},\zeta^{2},\zeta^{6},\zeta^{4},\zeta^{5},\zeta^{1}$ respectively, which is exactly the six elements $\sigma_{3},\sigma_{2},\sigma_{6},\sigma_{4},\sigma_{5},\sigma_{1}$ of $\operatorname{Gal}(K/\QQ)$. Hence $\langle\sigma\rangle=\operatorname{Gal}(K/\QQ)$.
\end{proof}
\noindent\textbf{(b)}
\begin{proof}
Between them, $\alpha$ and $\beta$ list each of $\zeta,\zeta^{2},\dots,\zeta^{6}$ exactly once, so
\[
\alpha+\beta=\zeta+\zeta^{2}+\cdots+\zeta^{6}=-1,
\]
because $\zeta$ is a root of $\Phi_7=x^{6}+x^{5}+\cdots+x+1$ and $\zeta\neq1$.

Expanding $\alpha\beta$ gives nine products, with exponents reduced modulo $7$:
\[
\zeta^{4},\ \zeta^{6},\ 1,\ \zeta^{5},\ 1,\ \zeta,\ 1,\ \zeta^{2},\ \zeta^{3}.
\]
Three of the nine are $\zeta^{7}=1$, and the other six are exactly $\alpha+\beta$. Hence
\[
\alpha\beta=3+(\alpha+\beta)=3-1=2.
\]

Since $\alpha+\beta=-1$ and $\alpha\beta=2$, the monic quadratic with roots $\alpha$ and $\beta$ is
\[
(x-\alpha)(x-\beta)=x^{2}-(\alpha+\beta)x+\alpha\beta=x^{2}+x+2.
\]
The quadratic formula with $a=1$, $b=1$, $c=2$ gives
\[
x=\frac{-1\pm\sqrt{1-8}}{2}=\frac{-1\pm\sqrt{-7}}{2},
\]
so the two roots are $\frac{-1+\sqrt{-7}}{2}$ and $\frac{-1-\sqrt{-7}}{2}$. Either may be called $\alpha$; taking $\alpha$ to be the root with positive imaginary part matches the observation that the sines in $\alpha=\zeta+\zeta^{2}+\zeta^{4}$ have positive sum. Since $\sqrt{-7}\notin\QQ$, the polynomial is irreducible and $\QQ(\alpha)=\QQ(\sqrt{-7})$ has degree $2$ over $\QQ$.

Finally, $\QQ(\sqrt{-7})$ is the only subfield of $K$ of degree $2$ over $\QQ$. By 14(a), $G=\operatorname{Gal}(K/\QQ)=\langle\sigma\rangle$ is cyclic of order $6$, and a cyclic group has at most one subgroup of a given order. So $G$ has exactly one subgroup of order $3$, namely $\{e,\sigma^{2},\sigma^{4}\}$. A subfield $L$ with $\QQ\subseteq L\subseteq K$ and $[L:\QQ]=2$ corresponds by the Galois correspondence to a subgroup $H\le G$ with $[G:H]=2$, that is $|H|=3$. There is only one such $H$, so there is only one such field, and $\QQ(\sqrt{-7})$ is it.
\end{proof}
\noindent\textbf{(c)}
\begin{proof}
The six values $\sigma^{k}(\eta)$, with exponents reduced modulo $7$, are
\[
\sigma^{0}(\eta)=\zeta+\zeta^{6},\quad
\sigma^{1}(\eta)=\zeta^{3}+\zeta^{4},\quad
\sigma^{2}(\eta)=\zeta^{2}+\zeta^{5},
\]
\[
\sigma^{3}(\eta)=\zeta^{6}+\zeta,\quad
\sigma^{4}(\eta)=\zeta^{4}+\zeta^{3},\quad
\sigma^{5}(\eta)=\zeta^{5}+\zeta^{2}.
\]
The rule behind them is $\sigma^{k}(\zeta)=\zeta^{3^{k}}$ with $3^{k}$ then reduced modulo $7$: $3^{3}=27\equiv6$, so $\sigma^{3}(\zeta)=\zeta^{6}$. Since $\sigma$ fixes $\QQ$ and respects inverses, $\sigma^{k}(\eta)=\sigma^{k}(\zeta)+(\sigma^{k}(\zeta))^{-1}$.

The list collapses to three elements, each occurring twice: $\eta=\zeta+\zeta^{6}$ at $k=0,3$; $\zeta^{3}+\zeta^{4}$ at $k=1,4$; $\zeta^{2}+\zeta^{5}$ at $k=2,5$. So the stabilizer of $\eta$ in $G$ is $\{\sigma^{0},\sigma^{3}\}$, of order $2$, and $[\QQ(\eta):\QQ]=6/2=3$. The three distinct values are the conjugates of $\eta$.

The three conjugates between them contain each of $\zeta,\zeta^{2},\dots,\zeta^{6}$ exactly once, so
\[
\eta+\sigma(\eta)+\sigma^{2}(\eta)=\zeta+\zeta^{2}+\cdots+\zeta^{6}=-1,
\]
because $\zeta$ is a root of $\Phi_7=x^{6}+\cdots+x+1$ and $\zeta\ne1$. The sum of the roots is $-1$, so the coefficient of $x^{2}$ in the cubic is $1$.

The first pairwise product is
\[
\eta\,\sigma(\eta)=(\zeta+\zeta^{6})(\zeta^{3}+\zeta^{4})=\zeta^{4}+\zeta^{5}+\zeta^{9}+\zeta^{10}=\zeta^{2}+\zeta^{3}+\zeta^{4}+\zeta^{5},
\]
four distinct powers, so no terms combine.

The second is
\[
\eta\,\sigma^{2}(\eta)=(\zeta+\zeta^{6})(\zeta^{2}+\zeta^{5})=\zeta^{3}+\zeta^{6}+\zeta^{8}+\zeta^{11}=\zeta+\zeta^{3}+\zeta^{4}+\zeta^{6},
\]
and the third is
\[
\sigma(\eta)\,\sigma^{2}(\eta)=(\zeta^{3}+\zeta^{4})(\zeta^{2}+\zeta^{5})=\zeta^{5}+\zeta^{8}+\zeta^{6}+\zeta^{9}=\zeta+\zeta^{2}+\zeta^{5}+\zeta^{6}.
\]
Adding the three products term by term, each of $\zeta,\zeta^{2},\dots,\zeta^{6}$ occurs exactly twice, so
\[
\eta\sigma(\eta)+\eta\sigma^{2}(\eta)+\sigma(\eta)\sigma^{2}(\eta)=2(\zeta+\zeta^{2}+\cdots+\zeta^{6})=-2,
\]
and the coefficient of $x$ is $-2$.

The product of all three is
\[
\eta\eta_1\eta_2=(\zeta+\zeta^{6})(\zeta^{5}+\zeta+\zeta^{6}+\zeta^{2})
=\zeta^{6}+\zeta^{2}+1+\zeta^{3}+\zeta^{4}+1+\zeta^{5}+\zeta
=2+(\zeta+\zeta^{2}+\cdots+\zeta^{6})=2-1=1.
\]
Two of those eight products are $\zeta^{7}=1$; the other six are exactly $\zeta,\zeta^{2},\dots,\zeta^{6}$, whose sum is $-1$. So the product is $1$ and the constant term is $-1$. The minimal polynomial is therefore
\[
x^{3}-(\eta+\sigma(\eta)+\sigma^{2}(\eta))x^{2}+(\eta\sigma(\eta)+\eta\sigma^{2}(\eta)+\sigma(\eta)\sigma^{2}(\eta))x-\eta\sigma(\eta)\sigma^{2}(\eta)
=x^{3}+x^{2}-2x-1.
\]

It remains to show that $\QQ(\eta)$ is the only cubic subfield. Let $L\subseteq K$ be a subfield with $[L:\QQ]=3$, and let $N$ be the subgroup of $G$ that fixes $L$. Then
\[
[L:\QQ]=[G:N]=3.
\]
There is only one subgroup of $G$ of index $3$, namely $H=\langle\sigma^{3}\rangle=\{\sigma_1,\sigma_6\}$, so $N=H$ and $L=K^{H}=\QQ(\eta)$. Hence every cubic subfield of $K$ is $\QQ(\eta)$.
\end{proof}

\begin{problem}{15}
  \textit{The real subfield of a cyclotomic field}\par

\noindent\textbf{(b)}
\begin{proof}
Write $s=\zeta+\zeta^{-1}$. Since $\zeta^{-1}\zeta=1$, substituting $x=\zeta$ gives
\[
\zeta^{2}-s\zeta+1=\zeta^{2}-\zeta^{2}-\zeta^{-1}\zeta+1=0.
\]
Both coefficients $s$ and $1$ lie in $K^{+}=\QQ(s)$, so this is a monic quadratic over $K^{+}$ with $\zeta$ as a root. Its coefficients are real as well: $s=2\cos(2\pi/n)$ is real, and every element of $K^{+}=\QQ(s)$ is a rational function of $s$ with rational coefficients, hence real.

So every element of $K^{+}$ is real, while $\zeta\notin\RR$ for $n\ge3$. Hence $\zeta\notin K^{+}$. Let $m$ be the minimal polynomial of $\zeta$ over $K^{+}$. Then $m$ divides $x^{2}-sx+1$, so $\deg m\le2$, and $\deg m\ne1$ because a root of a linear polynomial over $K^{+}$ lies in $K^{+}$. Therefore $\deg m=2$, with $m=x^{2}-sx+1$, and $[K:K^{+}]=2$.

For the fixed field, $c$ fixes every real number and every rational number, in particular the generator $s$, so $c$ fixes all of $K^{+}=\QQ(s)$; that gives $K^{+}\subseteq K^{\langle c\rangle}$. Conversely, $K^{+}$ contains the coefficients of $m$, so $\zeta$ satisfies a monic quadratic over $K^{\langle c\rangle}$ and $[K:K^{\langle c\rangle}]\le2$. That index is not $1$: if $K^{\langle c\rangle}=K$ then $c$ fixes every element of $K$, which is what being the identity means, but $c(\zeta)=\zeta^{-1}\ne\zeta$ because $\zeta^{2}\ne1$ for $n\ge3$. The tower $K^{+}\subseteq K^{\langle c\rangle}\subseteq K$ now gives
\[
2=[K:K^{+}]=[K:K^{\langle c\rangle}]\,[K^{\langle c\rangle}:K^{+}],
\]
and the first factor is $2$, so the second is $1$ and $K^{\langle c\rangle}=K^{+}$.

Finally $K^{+}=K\cap\RR$, by two inclusions. The inclusion $K^{+}\subseteq K\cap\RR$ is the one that needs $s=2\cos(2\pi/n)$ to be real: $K^{+}=\QQ(s)\subseteq\RR$, since a real generator and rational coefficients give a real field, while $K^{+}\subseteq K$ is plain. The inclusion $K\cap\RR\subseteq K^{+}$ needs only that $c$ fixes real numbers: an $x\in K\cap\RR$ lies in $K$ and satisfies $c(x)=x$, so $x\in K^{\langle c\rangle}=K^{+}$. Hence $K^{+}=K\cap\RR$.
\end{proof}

\end{proof}
\noindent\textbf{(d)}
\begin{proof}
With $n=5$, $\zeta^{-1}=\zeta^{4}$, so
\[
s=\zeta+\zeta^{-1}=\zeta+\zeta^{4}=2\cos(2\pi/5).
\]
Squaring,
\[
s^{2}=\zeta^{2}+2\zeta^{5}+\zeta^{8}=\zeta^{2}+2+\zeta^{3},
\]
so $\zeta^{3}+\zeta^{2}=s^{2}-1$, while $\zeta^{4}+\zeta=s$. The relation $\zeta^{4}+\zeta^{3}+\zeta^{2}+\zeta+1=0$ therefore becomes
\[
s+(s^{2}-1)=0,
\]
that is $s^{2}+s-1=0$. Its roots are $\tfrac{-1\pm\sqrt5}{2}$.

The element $s$ is the plus root. Since $2\pi/5$ lies in the first quadrant, $2\cos(2\pi/5)>0$, so $s>0$. The minus root $\tfrac{-1-\sqrt5}{2}$ is negative, so it cannot be $s$. Hence
\[
s=\frac{-1+\sqrt5}{2},\qquad \cos(2\pi/5)=\frac{s}{2}=\frac{\sqrt5-1}{4}.
\]

For the field, $s=\tfrac{-1+\sqrt5}{2}\in\QQ(\sqrt5)$, so $K^{+}=\QQ(s)\subseteq\QQ(\sqrt5)$. Conversely $\sqrt5=2s+1$, and $2,1\in\QQ$, so $\sqrt5\in K^{+}$ and $\QQ(\sqrt5)\subseteq K^{+}$. Therefore $K^{+}=\QQ(\sqrt5)$.
\end{proof}

\begin{problem}{16}
  \textit{Being Galois is not transitive}\par
  Let $\alpha=\sqrt[4]{2}\in\RR$.
  \begin{enumerate}[label=(\alph*)]
    \item Show that $[\QQ(\alpha):\QQ]=4$ and that $\QQ(\sqrt2)$ is an intermediate field.
    \item Show that $\QQ(\sqrt2)/\QQ$ and $\QQ(\alpha)/\QQ(\sqrt2)$ are both Galois.
    \item Show that $\QQ(\alpha)/\QQ$ is \emph{not} Galois, in two ways: by computing $\Aut(\QQ(\alpha)/\QQ)$, and by exhibiting an irreducible polynomial over $\QQ$ with a root in $\QQ(\alpha)$ that does not split there.
    \item Find the smallest extension of $\QQ(\alpha)$ that is Galois over $\QQ$, and compute its degree.
  \end{enumerate}
\end{problem}
% ===== SOLUTION 16 =====
\noindent\textbf{(a)}
\begin{proof}
The polynomial $x^{4}-2$ is irreducible over $\QQ$ by Eisenstein with $p=2$, so the minimal polynomial of $\alpha$ has degree $4$ and
\[
[\QQ(\alpha):\QQ]=4.
\]
Also $\sqrt[4]{2}\cdot\sqrt[4]{2}=\sqrt2\in\QQ(\alpha)$, so clearly
\[
\QQ(\sqrt2)\subseteq\QQ(\alpha).
\]
\end{proof}
\noindent\textbf{(b)}
\begin{proof}
\textbf{First, $\QQ(\sqrt2)$ over $\QQ$.} The polynomial $x^{2}-2$ has roots $\pm\sqrt2$, and
\[
\QQ(-\sqrt2,\sqrt2)=\QQ(\sqrt2)
\]
since $-\sqrt2\in\QQ(\sqrt2)$.

\textbf{Second, $\QQ(\alpha)$ over $\QQ(\sqrt2)$.} The polynomial $x^{2}-\sqrt2$ has roots $\sqrt[4]{2}$ and $-\sqrt[4]{2}$, which are $\alpha$ and $-\alpha$, and each lies in $\QQ(\alpha)$. Hence
\[
\QQ(\alpha,-\alpha)=\QQ(\alpha).
\]
\end{proof}
\noindent\textbf{(c)}
\begin{proof}
\textbf{First way.} The roots of $x^{4}-2$ are $\pm\sqrt[4]{2}$ and $\pm\sqrt[4]{2}\,i$, which is $\pm\alpha$ and $\pm\alpha i$. The first two lie in $\QQ(\alpha)$ and the last two do not. So
\[
|\operatorname{Gal}(\QQ(\alpha)/\QQ)|\le 2!=2,
\]
and
\[
2<4=[\QQ(\alpha):\QQ],
\]
so $\QQ(\alpha)/\QQ$ is not Galois.

\textbf{Second way.} $x^{4}-2\in\QQ[x]$, but $\pm\sqrt[4]{2}\,i=\pm\alpha i\notin\QQ(\alpha)$, so $\QQ(\alpha)/\QQ$ is not Galois.
\end{proof}

\noindent\textbf{(d)}
\begin{proof}
The splitting field of $x^{4}-2$ is $\QQ(\alpha,i)$: it is generated by $\alpha$ and $i$, and the four roots $\pm\alpha,\pm i\alpha$ of $x^{4}-2$ all lie in it. So $\QQ(\alpha,i)$ is Galois over $\QQ$, and what is left is to show that it is the smallest such field containing $\QQ(\alpha)$.

For the degree, the tower $\QQ\subseteq\QQ(\alpha)\subseteq\QQ(\alpha,i)$ gives
\[
[\QQ(\alpha,i):\QQ]=[\QQ(\alpha,i):\QQ(\alpha)]\,[\QQ(\alpha):\QQ]=2\cdot4=8.
\]
The second factor is $[\QQ(\alpha):\QQ]=4$ from 16(a). The first is $2$: the element $i$ satisfies $x^{2}+1\in\QQ(\alpha)[x]$, and $i\notin\QQ(\alpha)$ because $\QQ(\alpha)\subseteq\RR$ while $i\notin\RR$, so $x^{2}+1$ is the minimal polynomial of $i$ over $\QQ(\alpha)$.

The same total appears through $\QQ(i)$. Since $[\QQ(i):\QQ]=2$ with minimal polynomial $x^{2}+1$, the tower $\QQ\subseteq\QQ(i)\subseteq\QQ(\alpha,i)$ gives $[\QQ(\alpha,i):\QQ(i)]=8/2=4$, so this tower gives $4\cdot2=8$ as well.

Now let $L$ be a Galois extension of $\QQ$ with $\QQ(\alpha)\subseteq L$. The polynomial $x^{4}-2$ is irreducible over $\QQ$ by Eisenstein at $2$, so it is the minimal polynomial of $\alpha$ over $\QQ$. A Galois extension is normal, so $L$ contains every root in $\CC$ of the minimal polynomial over $\QQ$ of each of its elements. Applied to $\alpha$, this puts all four roots $\pm\alpha,\pm i\alpha$ of $x^{4}-2$ in $L$; in particular $i\alpha\in L$.

Also $\alpha\in L$ and $\alpha\ne0$, so $\alpha^{-1}\in L$, and a field is closed under multiplication, hence
\[
i=i\alpha\cdot\alpha^{-1}\in L.
\]
With $\alpha$ and $i$ both in $L$, we get $\QQ(\alpha,i)\subseteq L$. So every Galois extension of $\QQ$ containing $\QQ(\alpha)$ contains $\QQ(\alpha,i)$, and $\QQ(\alpha,i)$ is itself one. Therefore $\QQ(\alpha,i)$ is the smallest such extension, and its degree is $8$.
\end{proof}

\noindent\textbf{(a)}
\begin{proof}
For existence, $\alpha$ and $i\alpha$ are both roots of $x^{4}-2$. Put $K=\QQ(i)$, so $E=K(\alpha)$ is the splitting field of $x^{4}-2$ over $K$ and
\[
[K(\alpha):K]=\frac{[E:\QQ]}{[\QQ(i):\QQ]}=\frac{8}{2}=4.
\]
The polynomial $x^{4}-2$ is monic of degree $4$ over $K$ and vanishes at $\alpha$, so it is the minimal polynomial of $\alpha$ over $K$, hence irreducible there. Corollary 2 of Theorem 7.5 gives an automorphism $\sigma$ of $E$ fixing $K$ with $\sigma(\alpha)=i\alpha$. It fixes $i$ and it fixes $\QQ$, so it lies in $\operatorname{Gal}(E/\QQ)$.

Now $\sigma^{4}(\alpha)=i^{4}\alpha=\alpha$ and $\sigma^{4}(i)=i$, and $\alpha$ and $i$ generate $E$, so $\sigma^{4}=e$. Since
\[
\sigma^{2}(\alpha)=i^{2}\alpha=-\alpha,
\]
$\sigma^{2}\ne e$, and so $|\sigma|=4$.

Complex conjugation $\tau$ satisfies $\tau^{2}=e$, and $\tau\ne e$ because $\tau(i)=-i$; hence $|\tau|=2$. As $\tau^{-1}=\tau$,
\[
\tau\sigma\tau^{-1}(i)=\tau\sigma(-i)=\tau(-i)=i=\sigma^{-1}(i),
\]
and
\[
\tau\sigma\tau^{-1}(i\alpha)=\tau\sigma(-i\alpha)=\tau\bigl(\sigma(-i)\sigma(\alpha)\bigr)=\tau\bigl((-i)(i\alpha)\bigr)=\tau(\alpha)=\alpha=\sigma^{-1}(i\alpha).
\]
The elements $i$ and $i\alpha$ generate $E$, because $\alpha=(i\alpha)/i$, so $\tau\sigma\tau^{-1}=\sigma^{-1}$.

A word in $\sigma$ and $\tau$ reduces, using $\tau\sigma\tau^{-1}=\sigma^{-1}$, to one of the eight forms $\sigma^{a}\tau^{b}$ with $0\le a<4$ and $0\le b<2$. These eight are distinct. If $\sigma^{a}\tau^{b}=\sigma^{a'}\tau^{b'}$, then evaluating at $i$ gives
\[
(-1)^{b}i=\sigma^{a}\tau^{b}(i)=\sigma^{a'}\tau^{b'}(i)=(-1)^{b'}i,
\]
so $b\equiv b'\pmod 2$; and evaluating at $\alpha$ gives
\[
i^{a}\alpha=\sigma^{a}\tau^{b}(\alpha)=\sigma^{a'}\tau^{b'}(\alpha)=i^{a'}\alpha,
\]
so $a\equiv a'\pmod 4$.

Finally, send $r$ to $\sigma$ and $s$ to $\tau$, where $D_{4}=\langle r,s\mid r^{4}=s^{2}=1,\ srs=r^{-1}\rangle$. The relations hold, so this is a homomorphism $D_{4}\to\operatorname{Gal}(E/\QQ)$, and its image contains the eight distinct elements above. Since $|D_{4}|=|\operatorname{Gal}(E/\QQ)|=8$, it is an isomorphism.
\end{proof}
\noindent\textbf{(b)}
\begin{proof}
For $H=\langle\sigma\rangle$: $\sigma(i)=i$, so $i$ is fixed by $\sigma$ and $\QQ(i)\subseteq E^{H}$.

For $H=\langle\sigma^{2},\tau\rangle$: an element is $\sigma^{2k}\tau^{j}$, and
\[
\sigma^{2k}\tau^{j}(\sqrt2)=\sigma^{2k}(\sqrt2)=\bigl(\sigma^{2k}(\alpha)\bigr)^{2}=(i^{2k}\alpha)^{2}=i^{4k}\sqrt2=\sqrt2,
\]
so $\sqrt2$ is fixed and $\QQ(\sqrt2)\subseteq E^{H}$.

For $H=\langle\sigma^{2},\sigma\tau\rangle$: an element is $\sigma^{2k}(\sigma\tau)^{j}=\sigma^{2k+j}\tau^{j}$ with $j=0$ or $1$, and
\[
\sigma^{2k+j}\tau^{j}(i\sqrt2)=(-1)^{j}i^{1+4k+2j}\sqrt2=i\sqrt2,
\]
because $i^{4k}=1$ and $i^{2j}=(-1)^{j}$, so the two signs cancel. Hence $i\sqrt2$ is fixed and $\QQ(i\sqrt2)=\QQ(\sqrt{-2})\subseteq E^{H}$.

Each $H$ has order $4$, so $[E:E^{H}]=|H|=4$ and therefore $[E^{H}:\QQ]=8/4=2$. Each of $\QQ(i)$, $\QQ(\sqrt2)$ and $\QQ(\sqrt{-2})$ has degree $2$ over $\QQ$, with minimal polynomials $x^{2}+1$, $x^{2}-2$ and $x^{2}+2$. A containment between fields of the same finite degree is an equality.
\end{proof}
