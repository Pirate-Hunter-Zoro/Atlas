\noindent\textbf{Standing assumptions.} From here on we assume the fundamental theorem: for a finite Galois extension $K/F$, the maps $H\mapsto K^H$ and $L\mapsto \operatorname{Gal}(K/L)$ are mutually inverse, inclusion-reversing bijections between the subgroups of $\operatorname{Gal}(K/F)$ and the intermediate fields of $K/F$, with $[K:K^H]=|H|$; and $L/F$ is Galois exactly when $\operatorname{Gal}(K/L)$ is normal in $\operatorname{Gal}(K/F)$, in which case $\operatorname{Gal}(L/F)\cong \operatorname{Gal}(K/F)/\operatorname{Gal}(K/L)$. As before, $\zeta_n=e^{2\pi i/n}$, and all fields are subfields of $\CC$ unless stated otherwise.

\begin{problem}{13}
  \textit{The subfields of $\QQ(\zeta_p)$}\par
  Let $p$ be an odd prime, $\zeta=\zeta_p$, $K=\QQ(\zeta)$ and $G=\operatorname{Gal}(K/\QQ)$.
  \begin{enumerate}[label=(\alph*)]
    \item Show that $G$ is cyclic of order $p-1$.
    \item Show that for each divisor $d$ of $p-1$ there is exactly one intermediate field $K_d$ with $[K_d:\QQ]=d$, and that every such $K_d$ is Galois over $\QQ$ with cyclic Galois group.
    \item Take $p=13$. Check that $2$ is a primitive root modulo $13$, and use it to list every subgroup of $(\ZZ/13\ZZ)^\times$ together with the degree over $\QQ$ of the corresponding fixed field.
    \item Let $H\le G$ be the subgroup of index $d$, viewed inside $(\ZZ/p\ZZ)^\times$, and set $\eta_H=\sum_{a\in H}\zeta^a$ (a \emph{Gaussian period}). Show that $K_d=\QQ(\eta_H)$. \emph{Hint:} $\zeta,\zeta^2,\dots,\zeta^{p-1}$ is a $\QQ$-basis of $K$.
  \end{enumerate}
\end{problem}
% ===== SOLUTION 13 =====
\noindent\textbf{(a)}
\begin{proof}
The minimal polynomial of $\zeta$ over $\QQ$ is
\[ \Phi_p(x)=\sum_{i=1}^{p}x^{i-1}=x^{p-1}+\dots+x+1, \]
whose $p-1$ roots are $\zeta,\zeta^2,\zeta^3,\dots,\zeta^{p-1}$. Each $\sigma\in G$ sends $\zeta$ to a root of $\Phi_p$, and for each root $\zeta^k$, $k\in\{1,\dots,p-1\}$, there is exactly one $\sigma_k\in G$ with $\sigma_k(\zeta)=\zeta^k$. So there are $p-1$ of them, and $|G|=p-1$.

Since $(\ZZ/p\ZZ)^\times$ is cyclic, there is a primitive root $g$ modulo $p$. The automorphism $\sigma_g$ with $\sigma_g(\zeta)=\zeta^g$ satisfies $\sigma_g^{\,j}(\zeta)=\zeta^{g^j}$, and in particular $\sigma_g^{\,p-1}(\zeta)=\zeta^{g^{p-1}}=\zeta$ by Fermat's little theorem.

Suppose $\sigma_k\in G$, with $\sigma_k(\zeta)=\zeta^k$. Since $g$ is a primitive root modulo $p$, there is an $n$ with $g^n\equiv k\pmod p$. So $\sigma_g^{\,n}(\zeta)=\zeta^{g^n}=\zeta^k=\sigma_k(\zeta)$, and since an element of $G$ is determined by its value at $\zeta$, $\sigma_g^{\,n}=\sigma_k$. Hence $\sigma_g$ generates $G$, and $G$ is cyclic of order $p-1$.
\end{proof}
\noindent\textbf{(b)}
\begin{proof}
For a subgroup $H\le G$ the index is $[G:H]=|G|/|H|$, so the subgroup of order $(p-1)/d$ has index $d$.

For $H\le G$ the fixed field is $K^H=\{k\in K:\sigma(k)=k \text{ for every }\sigma\in H\}$, and $[K:K^H]=|H|$.

A cyclic group has exactly one subgroup of each order dividing its order. Since $G$ is cyclic, it has exactly one subgroup $H$ of order $(p-1)/d$. By the correspondence $H$ has exactly one fixed field $K^H=K_d$, and by the index computation above $[K_d:\QQ]=d$.

Since $G$ is cyclic, every subgroup is normal, so $H\unlhd G$. Hence $K_d=K^H$ is Galois over $\QQ$, with Galois group $G/H$, a quotient of the cyclic group $G$ and therefore cyclic.
\end{proof}

\begin{problem}{15}
  \textit{The real subfield of a cyclotomic field}\par
  Let $n\ge3$, $\zeta=\zeta_n$, $K=\QQ(\zeta)$, and $K^+=\QQ(\zeta+\zeta^{-1})$.
  \begin{enumerate}[label=(\alph*)]
    \item Show that complex conjugation restricts to an automorphism $c$ of $K$ with $c(\zeta)=\zeta^{-1}$, and that $c$ has order $2$.
    \item Show that $\zeta$ is a root of $x^2-(\zeta+\zeta^{-1})x+1\in K^+[x]$, and deduce that $K^+=K^{\langle c\rangle}=K\cap\RR$ and $[K:K^+]=2$.
    \item Deduce that $[K^+:\QQ]=\varphi(n)/2$ and that $K^+=\QQ(\cos(2\pi/n))$.
    \item Take $n=5$. Show that $K^+=\QQ(\sqrt5)$, and deduce the exact value $\cos(2\pi/5)=\tfrac14(\sqrt5-1)$.
  \end{enumerate}
\end{problem}

\noindent\textbf{(b)}
\begin{proof}
Write $s=\zeta+\zeta^{-1}$. Since $\zeta^{-1}\zeta=1$, substituting $x=\zeta$ gives
\[
\zeta^{2}-s\zeta+1=\zeta^{2}-\zeta^{2}-\zeta^{-1}\zeta+1=0.
\]
Both coefficients $s$ and $1$ lie in $K^{+}=\QQ(s)$, so this is a monic quadratic over $K^{+}$ with $\zeta$ as a root. Its coefficients are real as well: $s=2\cos(2\pi/n)$ is real, and every element of $K^{+}=\QQ(s)$ is a rational function of $s$ with rational coefficients, hence real.

So every element of $K^{+}$ is real, while $\zeta\notin\RR$ for $n\ge3$. Hence $\zeta\notin K^{+}$. Let $m$ be the minimal polynomial of $\zeta$ over $K^{+}$. Then $m$ divides $x^{2}-sx+1$, so $\deg m\le2$, and $\deg m\ne1$ because a root of a linear polynomial over $K^{+}$ lies in $K^{+}$. Therefore $\deg m=2$, with $m=x^{2}-sx+1$, and $[K:K^{+}]=2$.

For the fixed field, $c$ fixes every real number and every rational number, in particular the generator $s$, so $c$ fixes all of $K^{+}=\QQ(s)$; that gives $K^{+}\subseteq K^{\langle c\rangle}$. Conversely, $K^{+}$ contains the coefficients of $m$, so $\zeta$ satisfies a monic quadratic over $K^{\langle c\rangle}$ and $[K:K^{\langle c\rangle}]\le2$. That index is not $1$: if $K^{\langle c\rangle}=K$ then $c$ fixes every element of $K$, which is what being the identity means, but $c(\zeta)=\zeta^{-1}\ne\zeta$ because $\zeta^{2}\ne1$ for $n\ge3$. The tower $K^{+}\subseteq K^{\langle c\rangle}\subseteq K$ now gives
\[
2=[K:K^{+}]=[K:K^{\langle c\rangle}]\,[K^{\langle c\rangle}:K^{+}],
\]
and the first factor is $2$, so the second is $1$ and $K^{\langle c\rangle}=K^{+}$.

Finally $K^{+}=K\cap\RR$, by two inclusions. The inclusion $K^{+}\subseteq K\cap\RR$ is the one that needs $s=2\cos(2\pi/n)$ to be real: $K^{+}=\QQ(s)\subseteq\RR$, since a real generator and rational coefficients give a real field, while $K^{+}\subseteq K$ is plain. The inclusion $K\cap\RR\subseteq K^{+}$ needs only that $c$ fixes real numbers: an $x\in K\cap\RR$ lies in $K$ and satisfies $c(x)=x$, so $x\in K^{\langle c\rangle}=K^{+}$. Hence $K^{+}=K\cap\RR$.
\end{proof}
\noindent\textbf{(c)}
\begin{proof}
The degree tower is $[K:\QQ]=[K:K^{+}][K^{+}:\QQ]$, with $[K:\QQ]=\varphi(n)$ and $[K:K^{+}]=2$, so
\[
[K^{+}:\QQ]=\frac{\varphi(n)}{2}.
\]
For the field, $K^{+}=\QQ(\zeta+\zeta^{-1})$ and $\zeta+\zeta^{-1}=2\cos(2\pi/n)$, and $2\in\QQ$, so
\[
\QQ(2\cos(2\pi/n))=\QQ(\cos(2\pi/n)).
\]
\end{proof}
\noindent\textbf{(d)}
\begin{proof}
With $n=5$, $\zeta^{-1}=\zeta^{4}$, so
\[
s=\zeta+\zeta^{-1}=\zeta+\zeta^{4}=2\cos(2\pi/5).
\]
Squaring,
\[
s^{2}=\zeta^{2}+2\zeta^{5}+\zeta^{8}=\zeta^{2}+2+\zeta^{3},
\]
so $\zeta^{3}+\zeta^{2}=s^{2}-1$, while $\zeta^{4}+\zeta=s$. The relation $\zeta^{4}+\zeta^{3}+\zeta^{2}+\zeta+1=0$ therefore becomes
\[
s+(s^{2}-1)=0,
\]
that is $s^{2}+s-1=0$. Its roots are $\tfrac{-1\pm\sqrt5}{2}$.

The element $s$ is the plus root. Since $2\pi/5$ lies in the first quadrant, $2\cos(2\pi/5)>0$, so $s>0$. The minus root $\tfrac{-1-\sqrt5}{2}$ is negative, so it cannot be $s$. Hence
\[
s=\frac{-1+\sqrt5}{2},\qquad \cos(2\pi/5)=\frac{s}{2}=\frac{\sqrt5-1}{4}.
\]

For the field, $s=\tfrac{-1+\sqrt5}{2}\in\QQ(\sqrt5)$, so $K^{+}=\QQ(s)\subseteq\QQ(\sqrt5)$. Conversely $\sqrt5=2s+1$, and $2,1\in\QQ$, so $\sqrt5\in K^{+}$ and $\QQ(\sqrt5)\subseteq K^{+}$. Therefore $K^{+}=\QQ(\sqrt5)$.
\end{proof}

\begin{problem}{16}
  \textit{Being Galois is not transitive}\par
  Let $\alpha=\sqrt[4]{2}\in\RR$.
  \begin{enumerate}[label=(\alph*)]
    \item Show that $[\QQ(\alpha):\QQ]=4$ and that $\QQ(\sqrt2)$ is an intermediate field.
    \item Show that $\QQ(\sqrt2)/\QQ$ and $\QQ(\alpha)/\QQ(\sqrt2)$ are both Galois.
    \item Show that $\QQ(\alpha)/\QQ$ is \emph{not} Galois, in two ways: by computing $\Aut(\QQ(\alpha)/\QQ)$, and by exhibiting an irreducible polynomial over $\QQ$ with a root in $\QQ(\alpha)$ that does not split there.
    \item Find the smallest extension of $\QQ(\alpha)$ that is Galois over $\QQ$, and compute its degree.
  \end{enumerate}
\end{problem}
% ===== SOLUTION 16 =====
\noindent\textbf{(a)}
\begin{proof}
The polynomial $x^{4}-2$ is irreducible over $\QQ$ by Eisenstein with $p=2$, so the minimal polynomial of $\alpha$ has degree $4$ and
\[
[\QQ(\alpha):\QQ]=4.
\]
Also $\sqrt[4]{2}\cdot\sqrt[4]{2}=\sqrt2\in\QQ(\alpha)$, so clearly
\[
\QQ(\sqrt2)\subseteq\QQ(\alpha).
\]
\end{proof}
\noindent\textbf{(b)}
\begin{proof}
\textbf{First, $\QQ(\sqrt2)$ over $\QQ$.} The polynomial $x^{2}-2$ has roots $\pm\sqrt2$, and
\[
\QQ(-\sqrt2,\sqrt2)=\QQ(\sqrt2)
\]
since $-\sqrt2\in\QQ(\sqrt2)$.

\textbf{Second, $\QQ(\alpha)$ over $\QQ(\sqrt2)$.} The polynomial $x^{2}-\sqrt2$ has roots $\sqrt[4]{2}$ and $-\sqrt[4]{2}$, which are $\alpha$ and $-\alpha$, and each lies in $\QQ(\alpha)$. Hence
\[
\QQ(\alpha,-\alpha)=\QQ(\alpha).
\]
\end{proof}
\noindent\textbf{(c)}
\begin{proof}
\textbf{First way.} The roots of $x^{4}-2$ are $\pm\sqrt[4]{2}$ and $\pm\sqrt[4]{2}\,i$, which is $\pm\alpha$ and $\pm\alpha i$. The first two lie in $\QQ(\alpha)$ and the last two do not. So
\[
|\operatorname{Gal}(\QQ(\alpha)/\QQ)|\le 2!=2,
\]
and
\[
2<4=[\QQ(\alpha):\QQ],
\]
so $\QQ(\alpha)/\QQ$ is not Galois.

\textbf{Second way.} $x^{4}-2\in\QQ[x]$, but $\pm\sqrt[4]{2}\,i=\pm\alpha i\notin\QQ(\alpha)$, so $\QQ(\alpha)/\QQ$ is not Galois.
\end{proof}
