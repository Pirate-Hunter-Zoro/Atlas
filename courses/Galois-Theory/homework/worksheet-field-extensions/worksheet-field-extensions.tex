\begin{problem}{3(a)}
  \emph{Degrees.} Let $K/F$ be a finite extension of degree $n$. Show that for
  any $\alpha \in K$, the degree of $m_{\alpha,F}$ divides $n$.
\end{problem}

% ===== SOLUTION 3(a) =====
% Transcribed from the student's slate, 2026-09-18 (page t0054-r51, card 0061),
% with the case they worked first on t0054-r50 (card 0060) as the special case.
% Same three steps, same order, nothing added to the route.
% One reason-correction, made here rather than left to the student: the page
% says [F(alpha):F] "must divide [K:F] by the tower law". The tower law gives an
% equation, not a divisibility; the quotient is [K:F(alpha)], which is a
% positive integer. The write-up states the equation and names the quotient.
% The page also uses m_{alpha,F} without saying why it exists, so the existence
% line from the finiteness of [K:F] is stated first.
\begin{solution}
  Let $[K:F] = n$ and let $\alpha \in K$.

  Since $\degree{K}{F} = n$ is finite, the $n+1$ elements
  $1, \alpha, \dots, \alpha^{n}$ cannot be linearly independent over $F$, so
  $\alpha$ is algebraic over $F$ and $m_{\alpha,F}$ exists.

  Consider $\degree{F(\alpha)}{F}$. Since $\alpha \in K$ and $F \subseteq K$, we
  have $F \subseteq F(\alpha) \subseteq K$, so the tower law gives
  \[
    n = \degree{K}{F}
      = \degree{K}{F(\alpha)}\,\degree{F(\alpha)}{F} .
  \]
  Here $\degree{K}{F(\alpha)}$ is a positive integer, so it is a witness $q$
  with $n = q \cdot \degree{F(\alpha)}{F}$; that is,
  $\degree{F(\alpha)}{F}$ divides $n$.

  Finally $\deg m_{\alpha,F} = \degree{F(\alpha)}{F}$, so $\deg m_{\alpha,F}$
  divides $n$.

  \medskip
  \textbf{The special case first worked.} With $F = \mathbb{Q}$,
  $K = \mathbb{Q}(\sqrt[4]{2})$ and $\alpha = \sqrt{2}$, take
  $M = \mathbb{Q}(\sqrt{2})$: then $\degree{M}{\mathbb{Q}} = 2$ divides
  $4 = \degree{\mathbb{Q}(\sqrt[4]{2})}{\mathbb{Q}}$. Nothing in that argument
  used $\sqrt[4]{2}$ or the number $4$, which is why $F(\alpha)$ in place of
  $M$ runs in general.
\end{solution}

\begin{problem}{3(c)}
  Show that $\sqrt{2} \notin \mathbb{Q}(\sqrt[3]{2})$.
\end{problem}

% ===== SOLUTION 3(c) =====
% Transcribed from the student's slate, 2026-09-19 (page t0054-r54, card 0064).
% Same three steps in the same order: the degree of the minimal polynomial of
% sqrt(2), the degree of the extension, then 3(a) and 2 does not divide 3.
% Two reason-corrections, made here rather than left to the student:
%   - the page writes the minimal polynomial over the TOP field,
%     m_{alpha, Q(cbrt 2)}. 3(a) takes it over the bottom field, so the
%     write-up says m_{alpha, Q}. Over the top field every degree is 1 and the
%     divisibility is vacuous.
%   - the page gives deg m_{sqrt2,Q} = 2 from "the polynomial is x^2-2". Monic
%     with sqrt(2) as a root only bounds the degree above by 2; degree 1 is
%     ruled out by the irrationality of sqrt(2). That was agreed on card 0063
%     and is stated here.
\begin{solution}
  First, $\deg m_{\sqrt{2},\mathbb{Q}} = 2$. The polynomial is $x^{2} - 2$: it
  is monic, has $\sqrt{2}$ as a root and lies in $\mathbb{Q}[x]$, so
  $\deg m_{\sqrt{2},\mathbb{Q}} \leq 2$. Degree $1$ would make $\sqrt{2}$ a root
  of some $x - c$ with $c \in \mathbb{Q}$, that is $\sqrt{2} \in \mathbb{Q}$,
  and $\sqrt{2}$ is irrational. So the degree is exactly $2$.

  Further, $\degree{\mathbb{Q}(\sqrt[3]{2})}{\mathbb{Q}} = 3$, since
  $x^{3} - 2$ is irreducible over $\mathbb{Q}$.

  So by 3(a), applied to $K = \mathbb{Q}(\sqrt[3]{2})$ and $F = \mathbb{Q}$ with
  $n = 3$: for every $\alpha \in \mathbb{Q}(\sqrt[3]{2})$,
  $\deg m_{\alpha,\mathbb{Q}}$ divides $3$.

  So $\sqrt{2}$ cannot be in $\mathbb{Q}(\sqrt[3]{2})$, because
  $2 \nmid 3$.
\end{solution}
% ===== END SOLUTION 3(c) =====

\begin{problem}{6(a)}
  Let $K = \mathbb{Q}(\sqrt{2},\sqrt{3})$ and $\gamma = \sqrt{2} + \sqrt{3}$.
  Show that $\degree{K}{\mathbb{Q}} = 4$.
\end{problem}

% ===== SOLUTION 6(a) =====
% Transcribed from the student's slate, 2026-09-18 (page t0054-r42, turns
% leading to cards 0047-0050). Typeset exactly as written: same steps, same
% order, nothing added.
% Two reason-corrections, both agreed on cards 0048 and 0049 and made here
% rather than left to the student:
%   - the page justifies ab = 0 with "since a,b in Q"; the reason is that
%     sqrt(2) is irrational.
%   - the page flags each branch with "since b in Q" / "since a in Q"; the
%     contradictions are the irrationality of sqrt(6) and of sqrt(3).
\begin{solution}
  First, $\sqrt{3} \notin \mathbb{Q}(\sqrt{2})$.

  Suppose $\sqrt{3} = a + b\sqrt{2}$ with $a, b \in \mathbb{Q}$. Squaring,
  \[
    3 = a^{2} + 2ab\sqrt{2} + 2b^{2},
  \]
  so $2ab\sqrt{2} = 3 - a^{2} - 2b^{2}$ is rational, and since $\sqrt{2}$ is
  irrational this forces $ab = 0$.

  \emph{Suppose $a = 0$.} Then $\sqrt{3} = b\sqrt{2}$, so $2b = \sqrt{6}$. With
  $b \in \mathbb{Q}$ that makes $\sqrt{6}$ rational, and it is not.

  \emph{Suppose $b = 0$.} Then $a = \sqrt{3}$. With $a \in \mathbb{Q}$ that
  makes $\sqrt{3}$ rational, and it is not.

  So $\sqrt{3} \neq a + b\sqrt{2}$ for any $a, b \in \mathbb{Q}$, that is,
  $\sqrt{3} \notin \mathbb{Q}(\sqrt{2})$.

  \medskip
  \textbf{Note.} $\degree{\mathbb{Q}(\sqrt{2})}{\mathbb{Q}} = 2$, since
  $\mathbb{Q}(\sqrt{2}) = \mathbb{Q}[\sqrt{2}]$ has basis $\set{1, \sqrt{2}}$
  over $\mathbb{Q}$.

  \medskip
  \textbf{Further.} $\degree{K}{\mathbb{Q}(\sqrt{2})} = 2$, since
  $K = \mathbb{Q}(\sqrt{2})[\sqrt{3}]$ has basis $\set{1, \sqrt{3}}$ over
  $\mathbb{Q}(\sqrt{2})$: the minimal polynomial of $\sqrt{3}$ there divides
  $x^{2} - 3$, so has degree $2$ or $1$, and degree $1$ is ruled out by
  $\sqrt{3} \notin \mathbb{Q}(\sqrt{2})$.

  \medskip
  So by the tower law
  \[
    \degree{K}{\mathbb{Q}}
      = \degree{K}{\mathbb{Q}(\sqrt{2})}\,
        \degree{\mathbb{Q}(\sqrt{2})}{\mathbb{Q}}
      = 2 \cdot 2 = 4 .
  \]
\end{solution}

\begin{problem}{6(b)}
  With $K$ and $\gamma$ as above, show that $K = \mathbb{Q}(\gamma)$, and
  compute $m_{\gamma,\mathbb{Q}}$.
\end{problem}

% ===== SOLUTION 6(b) =====
% Transcribed from the student's slate, 2026-09-18 (pages t0054-r43, t0054-r44,
% t0054-r45 and t0054-r46, cards 0051-0054). Typeset exactly as written: same
% steps, same order, nothing added. Both halves are now agreed.
% One reason-correction, agreed on cards 0051 and 0053 and made here rather than
% left to the student: r43 justifies minimality with "degree 4, monic, so this
% is the minimal polynomial". Monic of degree 4 is not enough on its own. The
% reason is the degree count the student then gave on r46, and that is what the
% write-up says.
\begin{solution}
  $(\sqrt{2}+\sqrt{3})(\sqrt{3}-\sqrt{2}) = 3 - 2 = 1$, so
  $\sqrt{3}-\sqrt{2} = \gamma^{-1}$, and hence
  $\sqrt{3}-\sqrt{2} \in \mathbb{Q}(\gamma)$.

  Take $\gamma + \gamma^{-1} = 2\sqrt{3} \in \mathbb{Q}(\gamma)$. Since
  $2 \in \mathbb{Q}$, that puts $\sqrt{3} \in \mathbb{Q}(\gamma)$. So
  $\gamma - \sqrt{3} = \sqrt{2} \in \mathbb{Q}(\gamma)$.

  So $\mathbb{Q}(\gamma) = \mathbb{Q}(\sqrt{2},\sqrt{3})$. Both contain
  $\sqrt{2}+\sqrt{3}$, $\sqrt{2}$ and $\sqrt{3}$; $\mathbb{Q}(\gamma)$ is the
  smallest field extension of $\mathbb{Q}$ containing $\sqrt{2}+\sqrt{3}$, and
  $\mathbb{Q}(\sqrt{2},\sqrt{3})$ is the smallest field extension of
  $\mathbb{Q}$ containing $\sqrt{2}$ and $\sqrt{3}$. So they must be the same.

  \medskip
  \textbf{The minimal polynomial.} Squaring twice,
  \[
    \gamma^{2} = 5 + 2\sqrt{6}
    \implies (\gamma^{2}-5)^{2} = 24
    \implies \gamma^{4} - 10\gamma^{2} + 25 = 24
    \implies \gamma^{4} - 10\gamma^{2} + 1 = 0 .
  \]

  Now $\degree{\mathbb{Q}(\gamma)}{\mathbb{Q}} = 4$, since
  $\mathbb{Q}(\gamma) = \mathbb{Q}(\sqrt{2},\sqrt{3})$ and that degree is $4$ by
  (a). So $\deg m_{\gamma,\mathbb{Q}} = 4$, meaning $x^{4} - 10x^{2} + 1$ is the
  monic polynomial which is $m_{\gamma,\mathbb{Q}}$: any lower degree cannot have
  $\gamma$ as a root.
\end{solution}

\begin{problem}{6(c)}
  With $K$ and $\gamma$ as above, find three proper intermediate fields.
\end{problem}

% ===== SOLUTION 6(c) =====
% Transcribed from the student's slate, 2026-09-18 (page t0054-r47, card 0055).
% Typeset exactly as written: the three fields and, beside each, the elements of
% K it omits. Nothing added beyond naming what the page marks with arrows.
\begin{solution}
  $\degree{\mathbb{Q}(\gamma)}{\mathbb{Q}} = 4$, since
  $\mathbb{Q}(\gamma) = \mathbb{Q}(\sqrt{2},\sqrt{3})$.

  Intermediate fields of $\mathbb{Q}(\sqrt{2},\sqrt{3}) : \mathbb{Q}$:
  \[
    \begin{array}{lcl}
      \mathbb{Q}(\sqrt{2}) & \longrightarrow & \text{no } \sqrt{3},\ \sqrt{6}\\
      \mathbb{Q}(\sqrt{3}) & \longrightarrow & \text{no } \sqrt{2},\ \sqrt{6}\\
      \mathbb{Q}(\sqrt{6}) & \longrightarrow & \text{no } \sqrt{2},\ \sqrt{3}
    \end{array}
  \]

  Each is a proper intermediate field: each contains $\mathbb{Q}$ and sits
  inside $K$, each is bigger than $\mathbb{Q}$ because its generator is
  irrational, and none is all of $K$ because each omits the two elements of $K$
  named beside it. Those omissions also make the three fields distinct from one
  another.
\end{solution}
